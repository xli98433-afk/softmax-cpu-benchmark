cmake_minimum_required(VERSION 3.16)
project(softmax_perf_demo VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(softmax_lib src/softmax.cpp)
target_include_directories(softmax_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(softmax_lib PRIVATE /W4)
else()
    target_compile_options(softmax_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(softmax_tests tests/test_softmax.cpp)
target_link_libraries(softmax_tests PRIVATE softmax_lib)
add_executable(softmax_bench src/benchmark.cpp)
target_link_libraries(softmax_bench PRIVATE softmax_lib)
enable_testing()
add_test(NAME correctness_and_edge_cases COMMAND softmax_tests)
